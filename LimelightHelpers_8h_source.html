156 inline nt::DoubleArrayEntry& getLimelightDoubleArrayEntry(const std::string& tableName,
157 const std::string& entryName) {
158 const std::string key = tableName + "/" + entryName;
159 auto it = doubleArrayEntries.find(key);
160 if (it == doubleArrayEntries.end()) {
161 std::shared_ptr<nt::NetworkTable> table = getLimelightNTTable(tableName);
162 nt::DoubleArrayTopic daTopic = table->GetDoubleArrayTopic(entryName);
163 nt::DoubleArrayEntry entry = daTopic.GetEntry(std::span<double>{});
164 doubleArrayEntries.emplace(key, std::move(entry));
165 return doubleArrayEntries[key];
166 }
167 return it->second;
